cmake_minimum_required(VERSION 3.16)
project(ScholarDesk VERSION 1.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(scholar_core src/registry.cpp)
target_include_directories(scholar_core PUBLIC include)
add_executable(ScholarDesk src/main.cpp src/console.cpp)
target_link_libraries(ScholarDesk PRIVATE scholar_core)
if(WIN32)
  enable_language(RC)
  set_target_properties(ScholarDesk PROPERTIES OUTPUT_NAME ScholarDeskConsole)
  add_executable(ScholarDeskDesktop WIN32 src/desktop.cpp app.rc)
  target_include_directories(ScholarDeskDesktop PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  set_target_properties(ScholarDeskDesktop PROPERTIES OUTPUT_NAME ScholarDesk)
  target_link_libraries(ScholarDeskDesktop PRIVATE scholar_core comctl32 gdi32 shell32)
  if(MINGW)
    target_link_options(ScholarDeskDesktop PRIVATE -municode)
  endif()
  add_executable(desktop_tests WIN32 src/desktop.cpp app.rc)
  target_compile_definitions(desktop_tests PRIVATE SCHOLAR_DESKTOP_TESTS)
  target_include_directories(desktop_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(desktop_tests PRIVATE scholar_core comctl32 gdi32 shell32)
  if(MSVC)
    # app.rc already embeds our UAC/common-controls manifest as resource 1.
    target_link_options(ScholarDeskDesktop PRIVATE /MANIFEST:NO)
    target_link_options(desktop_tests PRIVATE /MANIFEST:NO)
  endif()
  if(MINGW)
    target_link_options(desktop_tests PRIVATE -municode)
  endif()
endif()
if(MSVC)
  target_compile_options(scholar_core PRIVATE /W4)
  target_compile_options(ScholarDesk PRIVATE /W4)
else()
  target_compile_options(scholar_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(ScholarDesk PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(scholar_tests tests/registry_tests.cpp)
target_link_libraries(scholar_tests PRIVATE scholar_core)
add_test(NAME registry_tests COMMAND scholar_tests)
if(WIN32)
  add_test(NAME desktop_tests COMMAND desktop_tests)
endif()
